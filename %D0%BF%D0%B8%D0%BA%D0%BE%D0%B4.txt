local function executeCode(code)
    if type(code) ~= "string" or code == "" then return end
    if type(loadstring) ~= "function" then
        warn("loadstring is unavailable in this environment.")
        return
    end
    local compileFunction, compileError = loadstring(code)
    if not compileFunction then
        warn(tostring(compileError))
        return
    end
    local ok, runError = pcall(compileFunction)
    if not ok then warn(tostring(runError)) end
end

local searchBusy = false
local function searchScripts(query)
    query = tostring(query or ""):gsub("^%s+", ""):gsub("%s+$", "")
    clearResults()
    if query == "" or searchBusy then return end

    searchBusy = true
    SearchButton.Text = "..."
    SearchButton.Active = false
    SearchBox.PlaceholderText = "SEARCHING..."

    local url = "https://scriptblox.com/api/script/search?q=" .. HttpService:UrlEncode(query) .. "&mode=free&page=1"
    local success, response = pcall(function()
        return game:HttpGet(url)
    end)

    if not success then
        warn("Search request failed: " .. tostring(response))
        SearchButton.Text = "FIND"
        SearchButton.Active = true
        SearchBox.PlaceholderText = "Script name..."
        searchBusy = false
        return
    end

    local decodeOK, data = pcall(function()
        return HttpService:JSONDecode(response)
    end)
    if not decodeOK or type(data) ~= "table" or type(data.result) ~= "table" or type(data.result.scripts) ~= "table" then
        warn("Could not parse the ScriptBlox response.")
        SearchButton.Text = "FIND"
        SearchButton.Active = true
        SearchBox.PlaceholderText = "Script name..."
        searchBusy = false
        return
    end

    for _, scriptData in ipairs(data.result.scripts) do
        addResult(scriptData)
    end
    SearchButton.Text = "FIND"
    SearchButton.Active = true
    SearchBox.PlaceholderText = "Script name..."
    searchBusy = false
end

SearchButton.MouseButton1Click:Connect(function()
    searchScripts(SearchBox.Text)
end)
SearchBox.FocusLost:Connect(function(enterPressed)
    if enterPressed then searchScripts(SearchBox.Text) end
end)

function Y:FindPlayer(query)
    if not query or query == "" then return nil end
    query = query:lower()

    if query == "me" then return LocalPlayer end
    if query == "random" then
        local list = {}
        for _, player in ipairs(Players:GetPlayers()) do
            if player ~= LocalPlayer and player.Character then
                table.insert(list, player)
            end
        end
        if #list == 0 then return nil end
        return list[math.random(1, #list)]
    end

    for _, player in ipairs(Players:GetPlayers()) do
        if player.Name:lower() == query then return player end
    end
    for _, player in ipairs(Players:GetPlayers()) do
        if player.Name:lower():find(query, 1, true) then return player end
    end
    return nil
end

function Y:Register(name, data)
    if type(name) ~= "string" or type(data) ~= "table" or type(data.Func) ~= "function" then return end
    data.Aliases = data.Aliases or {}
    self.Commands[name:lower()] = data
    for _, alias in ipairs(data.Aliases) do
        self.Aliases[alias:lower()] = name:lower()
    end
end

function Y:Get(name)
    name = (name or ""):lower()
    return self.Commands[name] or self.Commands[self.Aliases[name] or ""]
end

function Y:Run(input)
    if not input or input == "" then return end
    local args = {}
    for word in input:gmatch("%S+") do table.insert(args, word) end
    local name = table.remove(args, 1)
    if not name then return end
    local low = name:lower()

    if low:sub(1, 2) == "un" and #low > 2 then
        local base = self:Get(low:sub(3))
        if base and base.Unfunc then
            local ok, err = pcall(base.Unfunc, args, base)
            if not ok then notify("Error", tostring(err)) end
            return
        end
    end

    local command = self:Get(name)
    if not command then return notify(TITLE, "Unknown: " .. name) end
    local ok, err = pcall(command.Func, args, command)
    if not ok then notify("Error", tostring(err)) end
end
